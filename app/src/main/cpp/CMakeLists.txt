cmake_minimum_required(VERSION 3.22.1)
project(motionpin_rife LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=3 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(RIFE "${ROOT}/third_party/rife/src")
if(NOT EXISTS "${ROOT}/third_party/ncnn/glslang/CMakeLists.txt")
    message(FATAL_ERROR "Run scripts/setup-native.sh first to fetch verified native sources/models")
endif()
set(NCNN_VERSION 20220728 CACHE STRING "Pinned source commit date" FORCE)
foreach(FLAG NCNN_SHARED_LIB NCNN_OPENMP NCNN_BUILD_TESTS NCNN_BUILD_BENCHMARK NCNN_BUILD_TOOLS
             NCNN_BUILD_EXAMPLES NCNN_INSTALL_SDK NCNN_INT8 NCNN_BF16 NCNN_PIXEL_ROTATE
             NCNN_PIXEL_AFFINE NCNN_PIXEL_DRAWING NCNN_C_API NCNN_PYTHON)
    set(${FLAG} OFF CACHE BOOL "MotionPin build" FORCE)
endforeach()
set(NCNN_VULKAN ON CACHE BOOL "Required GPU path" FORCE)
set(NCNN_DISABLE_RTTI ON CACHE BOOL "Consistent native ABI" FORCE)
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "Propagate allocation failure" FORCE)
set(NCNN_ARM82 OFF CACHE BOOL "FP32 inference" FORCE)
set(NCNN_AVX OFF CACHE BOOL "Portable host test" FORCE)
set(NCNN_AVX2 OFF CACHE BOOL "Portable host test" FORCE)
set(NCNN_AVX512 OFF CACHE BOOL "Portable host test" FORCE)

# Only model-used operators are built; no image codec / desktop CLI dependency.
file(STRINGS "${RIFE}/CMakeLists.txt" upstream_layer_flags REGEX "option\\(WITH_LAYER_")
foreach(flag_line IN LISTS upstream_layer_flags)
    string(REGEX REPLACE ".*option\\((WITH_LAYER_[a-z0-9]+).* (ON|OFF)\\).*" "\\1;\\2" flag_parts "${flag_line}")
    list(GET flag_parts 0 flag_name)
    list(GET flag_parts 1 flag_value)
    set(${flag_name} ${flag_value} CACHE BOOL "RIFE operators" FORCE)
endforeach()
add_subdirectory("${ROOT}/third_party/ncnn" "${CMAKE_CURRENT_BINARY_DIR}/ncnn")

foreach(SHADER warp warp_pack4 warp_pack8)
    set(HEADER "${CMAKE_CURRENT_BINARY_DIR}/${SHADER}.comp.hex.h")
    add_custom_command(OUTPUT "${HEADER}"
        COMMAND ${CMAKE_COMMAND} -DSHADER_SRC=${RIFE}/${SHADER}.comp
            -DSHADER_COMP_HEADER=${HEADER} -P ${RIFE}/generate_shader_comp_header.cmake
        DEPENDS "${RIFE}/${SHADER}.comp" "${RIFE}/generate_shader_comp_header.cmake")
    list(APPEND SHADER_HEADERS "${HEADER}")
endforeach()
add_library(rife_engine STATIC rife_engine.cpp "${RIFE}/warp.cpp" ${SHADER_HEADERS})
target_include_directories(rife_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${RIFE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(rife_engine PRIVATE -fno-rtti)
target_link_libraries(rife_engine PUBLIC ncnn)
if(ANDROID)
    add_library(motionpin_rife SHARED native_rife.cpp)
    target_link_libraries(motionpin_rife PRIVATE rife_engine log)
    target_link_options(motionpin_rife PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
else()
    add_executable(rife_host_test host_test.cpp)
    target_link_libraries(rife_host_test PRIVATE rife_engine)
endif()
